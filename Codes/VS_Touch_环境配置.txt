#include <QHHeadersWin32.h>


int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow) {
	QHWin32* DisplayObject = new QHWin32; // 创建一个显示窗口
	DeviceSpace* Omni = new DeviceSpace; // 打开默认触摸
	DisplayObject->tell(Omni); // 告诉显示器全息存在
	Sphere* SimpleSphere = new Sphere; // 创建默认球体
	DisplayObject->tell(SimpleSphere); // 显示该球体的存在
	Cursor* OmniCursor = new Cursor; // 创建默认光标
	DisplayObject->tell(OmniCursor); // 告诉显示器存在一个光标
	qhStart(); //启动一切进程
}
